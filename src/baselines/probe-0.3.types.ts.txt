// 仅用于 Slint 实验；本版本接口与最终成品 API 会有很大差异。
export type UiProperties = Record<string, string | number | boolean>;

export interface PlaybackState {
  path: string | null;
  paused: boolean;
  position: number;
  duration: number | null;
  sessionId?: string;
  sourceKind?: "file" | "external" | "none";
  error?: string;
  gaussianEnabled: boolean;
  sigma: number;
}

export interface Effect {
  key: string;
  kind: "video" | "audio" | "subtitle";
  filter: string;
  files?: Record<string,string>;
  sessionId?: string;
}
export interface MediaTask {
  input: string;
  output: string;
  format: "clip" | "image" | "audio";
  videoFilter?: string;
  audioFilter?: string;
  start?: number;
  duration?: number;
  processing?: "current";
  processingSnapshot?: string;
  includeSubtitles?: boolean;
}
export interface JobState {
  id: string;
  state: "running" | "succeeded" | "failed" | "cancelled";
  error?: string;
  progressSeconds?: number;
}
export interface PluginContext {
  capabilities(): Promise<string[]>;
  effects: {
    set(effect: Effect): Promise<void>;
    remove(key: string): Promise<void>;
    list(): Promise<
      { owner: string; key: string; kind: string; filter: string }[]
    >;
    snapshot(options?: { includeSubtitles?: boolean }): Promise<{ id: string }>;
    releaseSnapshot(id: string): Promise<void>;
  };
  media: {
    start(task: MediaTask): Promise<JobState>;
    status(id: string): Promise<JobState>;
    cancel(id: string): Promise<void>;
  };
  sources: {
    open(options: {
      key: string;
      width: number;
      height: number;
      sampleRate?: number;
    }): Promise<{ sessionId: string }>;
    present(frame: {
      key: string;
      sessionId: string;
      rgb: string;
      pcm?: string;
    }): Promise<void>;
    close(key: string): Promise<void>;
    setPaused(options:{key:string;paused:boolean}):Promise<void>;
  };
  plugins: {
    validate(
      manifest: string,
    ): Promise<{ valid: boolean; diagnostics: string[] }>;
    install(manifest: string): Promise<{ id: string }>;
    remove(id: string): Promise<void>;
  };
  onInput(
    handler: (event: { key: string; pressed: boolean }) => void | Promise<void>,
  ): () => void;
  onAction(
    name: string,
    handler: (value: number) => void | Promise<void>,
  ): () => void;
  onPlayback(
    handler: (state: PlaybackState) => void | Promise<void>,
  ): () => void;
  ui: { patch(properties: UiProperties): Promise<void> };
  dialogs: {
    openMedia(): Promise<string | null>;
    openFile(options: {
      extensions: string[];
      title: string;
    }): Promise<string | null>;
    saveFile(options: {
      title: string;
      extensions: string[];
      suggestedName: string;
    }): Promise<string | null>;
  };
  player: {
    load(path: string): Promise<void>;
    togglePause(): Promise<void>;
    seek(seconds: number): Promise<void>;
    snapshot(): Promise<PlaybackState>;
    filters: {
      setGaussian(options: { enabled: boolean; sigma: number }): Promise<void>;
    };
  };
}

export interface PluginManifest {
  id: string;
  name: string;
  experimental: true;
  entry: string;
  commands?: { id: string; title: string; menu: string[] }[];
  // region 仅是此实验的挂载点约定，不是最终 placement API。
  ui: { file: string; component: string; global: string; region?: "sidebar" };
}

export type PluginDisposer = () => void | Promise<void>;
